# DB -- Postgres + pgvector via embedded pgserver, see docs/tech-stack.md #4.

# sqlalchemy.dialects.registry.load('postgresql'). psycopg2-binary is the
# real driver actually used; psycopg[binary] (v3) was listed here but never
# imported anywhere in app/, a real CI-only gap the local dev env masked
# (psycopg2-binary happened to already be installed there from before the
# serving/training requirements split).

pgvector
psycopg2-binary
